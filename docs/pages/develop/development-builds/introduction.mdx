import React, { useState, useEffect } from 'react';
import { 
  StyleSheet, Text, View, TouchableOpacity, ScrollView, 
  TextInput, Alert, Platform 
} from 'react-native';
// Librerías de Expo para PDF y Compartir
import * as Print from 'expo-print';
import * as Sharing from 'expo-sharing';
// Iconos nativos
import { Ionicons } from '@expo/vector-icons';
// Almacenamiento local nativo
import AsyncStorage from '@react-native-async-storage/async-storage';

// --- BASE DE DATOS DE PRUEBA ---
const DUMMY_CLIENTS = [
  { id: 'C001', razonSocial: 'PÚBLICO EN GENERAL', rfc: 'XAXX010101000', ciudad: 'LOCAL', direccion: 'CONOCIDO' },
  { id: 'C002', razonSocial: 'REFACCIONARIA EL MOTOR', rfc: 'MOT123456789', ciudad: 'CDMX', direccion: 'AV. CENTRAL 123' },
];

const DUMMY_PRODUCTS = [
  { sku: 'ACE-001', descripcion: 'ACEITE SINTÉTICO 5W-30', precioLista: 250 },
  { sku: 'FIL-002', descripcion: 'FILTRO DE AIRE ALTO FLUJO', precioLista: 150 },
  { sku: 'BUJ-003', descripcion: 'BUJÍA DE IRIDIUM NGK', precioLista: 90 },
];

export default function App() {
  const [activeTab, setActiveTab] = useState('inicio');
  
  // Estados de Datos
  const [selectedClient, setSelectedClient] = useState(DUMMY_CLIENTS[0]);
  const [cart, setCart] = useState([]);
  const [salesHistory, setSalesHistory] = useState([]);
  const [folioCounter, setFolioCounter] = useState(1);
  const [tempFolioInput, setTempFolioInput] = useState('1');

  // Filtros de Informes
  const [searchClient, setSearchClient] = useState('');
  const [searchDate, setSearchDate] = useState('');

  // Formateador de moneda
  const moneyFmt = (val) => '$' + parseFloat(val).toFixed(2).replace(/(\d)(?=(\d{3})+(?!\d))/g, '$1,');

  // Cargar datos al iniciar
  useEffect(() => {
    loadData();
  }, []);

  const loadData = async () => {
    try {
      const savedSales = await AsyncStorage.getItem('@sales_history');
      const savedFolio = await AsyncStorage.getItem('@folio_counter');
      if (savedSales) setSalesHistory(JSON.parse(savedSales));
      if (savedFolio) {
        setFolioCounter(parseInt(savedFolio));
        setTempFolioInput(savedFolio);
      }
    } catch (e) {
      console.log('Error loading data', e);
    }
  };

  const saveData = async (sales, folio) => {
    try {
      await AsyncStorage.setItem('@sales_history', JSON.stringify(sales));
      await AsyncStorage.setItem('@folio_counter', folio.toString());
    } catch (e) {
      console.log('Error saving data', e);
    }
  };

  // --- LÓGICA DEL CARRITO ---
  const addToCart = (product) => {
    const existing = cart.find(item => item.sku === product.sku);
    if (existing) {
      setCart(cart.map(item => item.sku === product.sku ? { ...item, qty: item.qty + 1 } : item));
    } else {
      setCart([...cart, { ...product, qty: 1, discount: 0 }]);
    }
    Alert.alert("Agregado", `${product.descripcion} agregado al carrito.`);
  };

  const clearCart = () => {
    Alert.alert("Vaciar Carrito", "¿Estás seguro?", [
      { text: "Cancelar", style: "cancel" },
      { text: "Vaciar", style: "destructive", onPress: () => setCart([]) }
    ]);
  };

  const getTotals = () => {
    let subtotal = 0;
    cart.forEach(item => {
      subtotal += (item.qty * item.precioLista) * (1 - (item.discount / 100));
    });
    return { subtotal, iva: subtotal * 0.16, total: subtotal * 1.16 };
  };

  // --- GENERACIÓN DE PDF NATIVO ---
  const generateNativePDF = async () => {
    if (cart.length === 0) {
      Alert.alert("Error", "El carrito está vacío");
      return;
    }

    const d = new Date();
    const dateStr = `${("0" + d.getDate()).slice(-2)}/${("0" + (d.getMonth() + 1)).slice(-2)}/${d.getFullYear()}`;
    const timeStr = `${("0" + d.getHours()).slice(-2)}:${("0" + d.getMinutes()).slice(-2)}`;
    const filterDateISO = `${d.getFullYear()}-${("0" + (d.getMonth() + 1)).slice(-2)}-${("0" + d.getDate()).slice(-2)}`;
    
    const currentFolio = `REM-${folioCounter.toString().padStart(4, '0')}`;
    const { subtotal, iva, total } = getTotals();

    // Guardar en el historial
    const newSale = {
      id: Date.now().toString(),
      folio: currentFolio,
      clientName: selectedClient.razonSocial,
      date: dateStr,
      time: timeStr,
      filterDate: filterDateISO,
      total: total,
      items: cart,
      status: 'PENDIENTE'
    };

    const updatedSales = [newSale, ...salesHistory];
    const newFolio = folioCounter + 1;
    
    setSalesHistory(updatedSales);
    setFolioCounter(newFolio);
    setTempFolioInput(newFolio.toString());
    saveData(updatedSales, newFolio);
    
    // Generar el HTML del recibo
    let rowsHtml = '';
    cart.forEach(item => {
      const rowT = (item.qty * item.precioLista) * (1 - (item.discount / 100));
      rowsHtml += `
        <tr>
          <td style="padding: 8px; border-bottom: 1px solid #ddd;">${item.sku}</td>
          <td style="padding: 8px; border-bottom: 1px solid #ddd;">${item.descripcion}</td>
          <td style="padding: 8px; border-bottom: 1px solid #ddd; text-align: center;">${item.qty}</td>
          <td style="padding: 8px; border-bottom: 1px solid #ddd; text-align: right;">${moneyFmt(rowT)}</td>
        </tr>
      `;
    });

    const htmlContent = `
      <html>
        <head>
          <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, minimum-scale=1.0, user-scalable=no" />
        </head>
        <body style="font-family: 'Helvetica Neue', Helvetica, Arial, sans-serif; padding: 20px; color: #333;">
          <div style="border-bottom: 2px solid #0055A4; padding-bottom: 10px; margin-bottom: 20px;">
            <h1 style="color: #0055A4; margin: 0; font-size: 28px;">REMISIÓN DE VENTA</h1>
            <h3 style="margin: 5px 0; color: #555;">Folio: <strong style="color: #000;">${currentFolio}</strong></h3>
            <p style="margin: 0; font-size: 12px; color: #777;">Fecha: ${dateStr} ${timeStr}</p>
          </div>
          <div style="margin-bottom: 20px;">
            <p style="margin: 0; font-weight: bold; font-size: 16px;">CLIENTE:</p>
            <p style="margin: 2px 0; font-size: 14px;">${selectedClient.razonSocial}</p>
            <p style="margin: 2px 0; font-size: 12px; color: #666;">RFC: ${selectedClient.rfc} | Dir: ${selectedClient.direccion}</p>
          </div>
          <table style="width: 100%; border-collapse: collapse; font-size: 12px; margin-bottom: 30px;">
            <tr style="background-color: #0055A4; color: white;">
              <th style="padding: 8px; text-align: left;">SKU</th>
              <th style="padding: 8px; text-align: left;">Descripción</th>
              <th style="padding: 8px; text-align: center;">Cant.</th>
              <th style="padding: 8px; text-align: right;">Importe</th>
            </tr>
            ${rowsHtml}
          </table>
          <div style="width: 100%; text-align: right;">
            <p style="margin: 5px 0; font-size: 14px;">Subtotal: <strong>${moneyFmt(subtotal)}</strong></p>
            <p style="margin: 5px 0; font-size: 14px;">IVA (16%): <strong>${moneyFmt(iva)}</strong></p>
            <h2 style="margin: 10px 0; color: #0055A4; font-size: 20px;">TOTAL: ${moneyFmt(total)}</h2>
          </div>
        </body>
      </html>
    `;

    try {
      // 1. Crear PDF en almacenamiento interno
      const { uri } = await Print.printToFileAsync({ html: htmlContent });
      
      // 2. Compartir nativamente (Abre WhatsApp, Drive, Correo, etc.)
      if (Platform.OS === 'android' || Platform.OS === 'ios') {
        await Sharing.shareAsync(uri, { UTI: '.pdf', mimeType: 'application/pdf' });
      }
      
      // Limpiar carrito después de compartir
      setCart([]);
      setActiveTab('informes');

    } catch (error) {
      Alert.alert('Error', 'No se pudo generar el PDF.');
      console.error(error);
    }
  };

  const updateFolioCounter = () => {
    const val = parseInt(tempFolioInput);
    if (!isNaN(val) && val > 0) {
      setFolioCounter(val);
      saveData(salesHistory, val);
      Alert.alert("Actualizado", `El contador iniciará en el folio ${val}`);
    } else {
      Alert.alert("Error", "Ingresa un número de folio válido");
    }
  };

  // --- RENDERIZADO DE PANTALLAS ---
  const renderInicio = () => (
    <ScrollView style={styles.screen}>
      <Text style={styles.sectionTitle}>Acciones Rápidas</Text>
      <View style={styles.grid}>
        <TouchableOpacity style={styles.gridCard} onPress={() => setActiveTab('cotizador')}>
          <Ionicons name="cart" size={32} color="#0055A4" />
          <Text style={styles.gridText}>Mostrador</Text>
        </TouchableOpacity>
        <TouchableOpacity style={styles.gridCard} onPress={() => setActiveTab('informes')}>
          <Ionicons name="stats-chart" size={32} color="#8b5cf6" />
          <Text style={styles.gridText}>Informes</Text>
        </TouchableOpacity>
      </View>
    </ScrollView>
  );

  const renderCotizador = () => {
    const { total } = getTotals();
    return (
      <View style={{ flex: 1 }}>
        <ScrollView style={styles.screen}>
          {/* Header Cotizador */}
          <View style={styles.quoteHeader}>
            <Text style={styles.quoteSubtitle}>Cotizando a:</Text>
            <Text style={styles.quoteTitle}>{selectedClient.razonSocial}</Text>
            <Text style={styles.quoteTotal}>Total: {moneyFmt(total)}</Text>
          </View>

          {/* Buscador Rápido (Simulado con botones para este MVP) */}
          <Text style={styles.sectionTitle}>Catálogo de Productos</Text>
          {DUMMY_PRODUCTS.map((prod) => (
            <TouchableOpacity key={prod.sku} style={styles.productCard} onPress={() => addToCart(prod)}>
              <View style={{ flex: 1 }}>
                <Text style={styles.prodSku}>{prod.sku}</Text>
                <Text style={styles.prodDesc}>{prod.descripcion}</Text>
              </View>
              <View style={styles.prodPriceBox}>
                <Text style={styles.prodPrice}>{moneyFmt(prod.precioLista)}</Text>
                <Ionicons name="add-circle" size={24} color="#0055A4" />
              </View>
            </TouchableOpacity>
          ))}

          <Text style={styles.sectionTitle}>En Carrito ({cart.length})</Text>
          {cart.length === 0 && <Text style={styles.emptyText}>El carrito está vacío</Text>}
          {cart.map((item, index) => (
            <View key={index} style={styles.cartCard}>
              <Text style={styles.cartSku}>{item.sku} - {item.descripcion}</Text>
              <Text style={styles.cartDetail}>Cant: {item.qty} | Precio: {moneyFmt(item.precioLista)}</Text>
            </View>
          ))}
          <View style={{ height: 100 }} /> 
        </ScrollView>

        {/* Botonera Flotante */}
        <View style={styles.actionFooter}>
          <TouchableOpacity style={[styles.btnAction, styles.btnDanger]} onPress={clearCart}>
            <Text style={styles.btnText}>Vaciar</Text>
          </TouchableOpacity>
          <TouchableOpacity style={[styles.btnAction, styles.btnPrimary, { flex: 2 }]} onPress={generateNativePDF}>
            <Ionicons name="share-social" size={18} color="#fff" />
            <Text style={[styles.btnText, { color: '#fff' }]}> Crear PDF</Text>
          </TouchableOpacity>
        </View>
      </View>
    );
  };

  const renderInformes = () => {
    // Filtrado de historial
    const filteredSales = salesHistory.filter(sale => {
      const matchClient = sale.clientName.toLowerCase().includes(searchClient.toLowerCase()) || sale.folio.toLowerCase().includes(searchClient.toLowerCase());
      const matchDate = searchDate ? sale.filterDate === searchDate : true;
      return matchClient && matchDate;
    });

    let totalDinero = filteredSales.reduce((acc, curr) => acc + curr.total, 0);

    return (
      <ScrollView style={styles.screen}>
        {/* Gestor de Folio */}
        <View style={styles.card}>
          <Text style={styles.label}>Configurar Siguiente Folio:</Text>
          <View style={styles.row}>
            <TextInput 
              style={styles.inputFlex} 
              keyboardType="number-pad"
              value={tempFolioInput}
              onChangeText={setTempFolioInput}
            />
            <TouchableOpacity style={styles.btnSmall} onPress={updateFolioCounter}>
              <Text style={{color: 'white', fontWeight: 'bold'}}>FIJAR</Text>
            </TouchableOpacity>
          </View>
        </View>

        {/* Totales */}
        <View style={styles.summaryCard}>
          <Text style={styles.summaryTitle}>Total Ventas Listadas</Text>
          <Text style={styles.summaryTotal}>{moneyFmt(totalDinero)}</Text>
          <Text style={styles.summarySubtitle}>{filteredSales.length} Remisiones</Text>
        </View>

        {/* Buscador y Filtros */}
        <View style={styles.card}>
          <Text style={styles.label}>Buscar Cliente o Folio:</Text>
          <TextInput 
            style={styles.input} 
            placeholder="Ej. Público en General"
            value={searchClient}
            onChangeText={setSearchClient}
          />
          <Text style={[styles.label, {marginTop: 10}]}>Filtrar por Fecha (YYYY-MM-DD):</Text>
          <TextInput 
            style={styles.input} 
            placeholder="Ej. 2026-06-15"
            value={searchDate}
            onChangeText={setSearchDate}
          />
          {(searchClient !== '' || searchDate !== '') && (
            <TouchableOpacity style={{marginTop: 10}} onPress={() => {setSearchClient(''); setSearchDate('')}}>
              <Text style={{color: 'red', textAlign: 'right', fontWeight: 'bold'}}>Limpiar Filtros</Text>
            </TouchableOpacity>
          )}
        </View>

        {/* Lista de Historial */}
        <Text style={styles.sectionTitle}>Historial de Remisiones</Text>
        {filteredSales.length === 0 && <Text style={styles.emptyText}>No hay registros.</Text>}
        {filteredSales.map((sale) => (
          <View key={sale.id} style={styles.historyCard}>
            <View style={styles.rowBetween}>
              <Text style={styles.historyFolio}>{sale.folio}</Text>
              <Text style={styles.historyDate}>{sale.date}</Text>
            </View>
            <Text style={styles.historyClient} numberOfLines={1}>{sale.clientName}</Text>
            <View style={[styles.rowBetween, { marginTop: 8 }]}>
              <Text style={styles.historyTotal}>{moneyFmt(sale.total)}</Text>
              {/* Aquí podrías agregar un botón para re-imprimir el PDF */}
            </View>
          </View>
        ))}
        <View style={{ height: 80 }} /> 
      </ScrollView>
    );
  };

  return (
    <View style={styles.container}>
      {/* App Bar */}
      <View style={styles.appBar}>
        <Text style={styles.appBarTitle}>Cotizador Nativo</Text>
      </View>

      {/* Main Content Area */}
      {activeTab === 'inicio' && renderInicio()}
      {activeTab === 'cotizador' && renderCotizador()}
      {activeTab === 'informes' && renderInformes()}

      {/* Bottom Navigation */}
      <View style={styles.bottomNav}>
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('inicio')}>
          <Ionicons name="home" size={24} color={activeTab === 'inicio' ? '#0055A4' : '#94a3b8'} />
          <Text style={[styles.navText, activeTab === 'inicio' && styles.navTextActive]}>Inicio</Text>
        </TouchableOpacity>
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('cotizador')}>
          <Ionicons name="cart" size={24} color={activeTab === 'cotizador' ? '#0055A4' : '#94a3b8'} />
          <Text style={[styles.navText, activeTab === 'cotizador' && styles.navTextActive]}>Carrito</Text>
        </TouchableOpacity>
        <TouchableOpacity style={styles.navItem} onPress={() => setActiveTab('informes')}>
          <Ionicons name="stats-chart" size={24} color={activeTab === 'informes' ? '#0055A4' : '#94a3b8'} />
          <Text style={[styles.navText, activeTab === 'informes' && styles.navTextActive]}>Informes</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
}
